// Consulta de estadísticas de proyectos y tecnologías vinculadas:

SELECT

p.id, p.nombre_proyecto, c.nombre_categoria, COUNT(t.tecnologia_id) AS total_tech

FROM

proyectos p

INNER JOIN

categorias c ON p.categoria_id = c.id

LEFT JOIN

proyecto_tecnologias t ON p.id = t.proyecto_id

GROUP BY

p.id, p.nombre_proyecto, c.nombre_categoria;
